44: // Kleines Rate-Limit ohne Zusatzpaket: max. N Anfragen pro Minute und IP
45: function rateLimit(max, windowMs = 60 * 1000, keyOf = (req) => req.ip) {
46:   const hits = new Map();
47:   setInterval(() => { const now = Date.now(); for (const [k, v] of hits) if (v.resetAt <= now) hits.delete(k); }, windowMs).unref();
48:   return (req, res, next) => {
49:     const now = Date.now();
50:     const key = keyOf(req);
51:     const h = hits.get(key);
52:     if (!h || h.resetAt <= now) { hits.set(key, { count: 1, resetAt: now + windowMs }); return next(); }
53:     h.count += 1;
54:     if (h.count > max) {
55:       res.set('Retry-After', String(Math.ceil((h.resetAt - now) / 1000)));
56:       return res.status(429).json({ error: 'Zu viele Anfragen. Bitte kurz warten.' });
57:     }
58:     next();
59:   };
60: }

106: function parseQuery(raw) {
107:   const tokens = raw.trim().replace(/#/g, ' ').split(/\s+/).filter(Boolean);
108:   if (tokens.length >= 2) {
109:     const last = tokens[tokens.length - 1];
110:     const m = last.match(/^([A-Za-z]{0,4})(\d{1,3})([A-Za-z]?)(?:\/([A-Za-z]{0,4}\d{1,3}))?$/);
111:     if (m) {
112:       return {
113:         name: tokens.slice(0, -1).join(' '),
114:         number: {
115:           prefix: m[1].toUpperCase(),
116:           digits: String(parseInt(m[2], 10)),
117:           suffix: m[3].toLowerCase(),
118:           total: m[4] ? String(parseInt(m[4].replace(/\D/g, ''), 10)) : null
119:         }
120:       };
121:     }
122:   }
123:   return { name: tokens.join(' '), number: null };
124: }

301:     return product ? cardmarket.applyProduct(card, product) : card;

314:     while (i < items.length) { const n = i++; out[n] = await fn(items[n]); }
