107: function parseQuery(raw) {
108:   const tokens = raw.trim().replace(/#/g, ' ').split(/\s+/).filter(Boolean);
109:   if (tokens.length >= 2) {
110:     const last = tokens[tokens.length - 1];
111:     const m = last.match(/^([A-Za-z]{0,4})(\d{1,3})([A-Za-z]?)(?:\/([A-Za-z]{0,4}\d{1,3}))?$/);
112:     if (m) {
113:       return {
114:         name: tokens.slice(0, -1).join(' '),
115:         number: {
116:           prefix: m[1].toUpperCase(),
117:           digits: String(parseInt(m[2], 10)),
118:           suffix: m[3].toLowerCase(),
119:           total: m[4] ? String(parseInt(m[4].replace(/\D/g, ''), 10)) : null
120:         }
121:       };
122:     }
123:   }
124:   return { name: tokens.join(' '), number: null };
125: }

127: function splitLocalId(localId) {
128:   const m = String(localId || '').match(/^([A-Za-z]*)(\d+)([A-Za-z]?)$/);
129:   if (!m) return null;
130:   return { prefix: m[1].toUpperCase(), digits: String(parseInt(m[2], 10)), suffix: m[3].toLowerCase() };
131: }

133: function matchesNumber(localId, number) {
134:   const parts = splitLocalId(localId);
135:   if (!parts) return false;
136:   return parts.digits === number.digits && parts.prefix === number.prefix && parts.suffix === (number.suffix || '');
137: }
