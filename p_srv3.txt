46: function rateLimit(max, windowMs = 60 * 1000, keyOf = (req) => req.ip) {
47:   const hits = new Map();
48:   setInterval(() => { const now = Date.now(); for (const [k, v] of hits) if (v.resetAt <= now) hits.delete(k); }, windowMs).unref();
49:   return (req, res, next) => {
50:     const now = Date.now();
51:     const key = keyOf(req);
52:     const h = hits.get(key);
53:     if (!h || h.resetAt <= now) { hits.set(key, { count: 1, resetAt: now + windowMs }); return next(); }
54:     h.count += 1;
55:     if (h.count > max) {
56:       res.set('Retry-After', String(Math.ceil((h.resetAt - now) / 1000)));
57:       return res.status(429).json({ error: 'Zu viele Anfragen. Bitte kurz warten.' });
58:     }
59:     next();
60:   };
61: }
